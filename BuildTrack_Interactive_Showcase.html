Core PyTorch Model Pipeline Implementation (`backend/vision_pipeline.py`)

import torch
import segmentation_models_pytorch as smp
import cv2
import numpy as np

class RoadVisionPipeline:
    def __init__(self):
        torch.set_num_threads(1) # Prevent CPU contention
        self.device = torch.device("cpu")
        self.model = smp.Unet(
            encoder_name="resnet18",
            encoder_weights=None,
            in_channels=3,
            classes=1
        )
        # Load trained road checkpoint (57.4 MB)
        ckpt = torch.load("backend/models/road_boundary_gravel_candidate.pth", map_location=self.device)
        self.model.load_state_dict(ckpt)
        self.model.eval()

    def segment_and_measure(self, image_np, gsd_m_per_px=0.0062):
        # 1. Preprocess frame
        tensor = torch.from_numpy(image_np.transpose((2, 0, 1))).float().unsqueeze(0) / 255.0
        with torch.no_grad():
            logits = self.model(tensor)
            prob_mask = torch.sigmoid(logits).squeeze().cpu().numpy()
        
        # 2. Extract road boundaries & lateral widths
        binary_mask = (prob_mask > 0.35).astype(np.uint8) * 255
        scanline_widths_px = [np.count_nonzero(row) for row in binary_mask[::10] if np.count_nonzero(row) > 0]
        mean_width_meters = float(np.mean(scanline_widths_px)) * gsd_m_per_px
        
        return {
            "mean_width_m": round(mean_width_meters, 2),
            "binary_mask": binary_mask,
            "confidence": float(np.mean(prob_mask[binary_mask > 0]))
        }
