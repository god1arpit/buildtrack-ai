import segmentation_models_pytorch as smp

# Strict single-thread execution to prevent cloud container OOM kills
torch.set_num_threads(1)
model = smp.Unet(encoder_name="resnet18", classes=1, in_channels=3)
model.load_state_dict(torch.load("road_boundary_gravel_candidate.pth", map_location="cpu"))
model.eval()

def compute_carriageway_widths(frame_bgr, gsd_m_per_px=0.0062):
    tensor = torch.from_numpy(frame_bgr.transpose((2, 0, 1))).float().unsqueeze(0) / 255.0
    with torch.no_grad():
        mask = (torch.sigmoid(model(tensor)).squeeze().numpy() > 0.35).astype(np.uint8) * 255
    widths_px = [np.count_nonzero(row) for row in mask[::10] if np.count_nonzero(row) > 0]
    return float(np.mean(widths_px)) * gsd_m_per_px
